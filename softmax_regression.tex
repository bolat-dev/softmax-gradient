\section*{Definitions}

\[
z_j = \sum_{m} W_{jm}\,x_m + b_j, \qquad
\hat{y}_i = \frac{e^{z_i}}{S}, \qquad
S = \sum_{k=1}^{C} e^{z_k}
\]

\[
L = -\sum_{i=1}^{C} y_i \ln \hat{y}_i, \qquad
y \text{ one-hot}, \quad \sum_i y_i = 1
\]

The summation indices ($k$, $i$, $m$) are dummy indices; $j$, the index we differentiate with respect to, is free.

\section*{Intermediate result}

\[
\ln \hat{y}_i = z_i - \ln S, \qquad
\frac{\partial S}{\partial z_j} = e^{z_j}
\quad\Longrightarrow\quad
\frac{\partial \ln S}{\partial z_j} = \frac{e^{z_j}}{S} = \hat{y}_j
\]

\section*{Derivative of $\ln \hat{y}_i$ with respect to $z_j$}

\[
\frac{\partial \ln \hat{y}_i}{\partial z_j}
= \frac{\partial z_i}{\partial z_j} - \frac{\partial \ln S}{\partial z_j}
= \delta_{ij} - \hat{y}_j
\]

This splits into two cases:

\[
\frac{\partial \ln \hat{y}_i}{\partial z_j} =
\begin{cases}
1 - \hat{y}_j & (i = j) \\
-\hat{y}_j & (i \neq j)
\end{cases}
\]

\section*{Applying it to the loss}

\[
\frac{\partial L}{\partial z_j}
= -\sum_i y_i \frac{\partial \ln \hat{y}_i}{\partial z_j}
= -\sum_i y_i \left(\delta_{ij} - \hat{y}_j\right)
\]

\[
= -\sum_i y_i \delta_{ij} + \hat{y}_j \sum_i y_i
= -y_j + \hat{y}_j \qquad \Big(\textstyle\sum_i y_i = 1\Big)
\]

\[
\boxed{\frac{\partial L}{\partial z_j} = \hat{y}_j - y_j}
\]

\section*{Gradient with respect to the parameters}

\[
\frac{\partial z_k}{\partial W_{jm}} = \delta_{kj}\,x_m, \qquad
\frac{\partial z_k}{\partial b_j} = \delta_{kj}
\]

\[
\frac{\partial L}{\partial W_{jm}}
= \sum_k \frac{\partial L}{\partial z_k}\frac{\partial z_k}{\partial W_{jm}}
= (\hat{y}_j - y_j)\,x_m
\]

\[
\frac{\partial L}{\partial b_j} = \hat{y}_j - y_j
\]

\section*{Batch of $N$ examples}

\[
L = \frac{1}{N}\sum_{n=1}^{N} L^{(n)}
\]

\[
\frac{\partial L}{\partial W_{jm}}
= \frac{1}{N}\sum_{n=1}^{N}\left(\hat{y}_j^{(n)} - y_j^{(n)}\right)x_m^{(n)},
\qquad
\frac{\partial L}{\partial b_j}
= \frac{1}{N}\sum_{n=1}^{N}\left(\hat{y}_j^{(n)} - y_j^{(n)}\right)
\]

In matrix form, for $X \in \mathbb{R}^{N\times M}$ and $\hat{Y}, Y \in \mathbb{R}^{N\times C}$:

\[
\frac{\partial L}{\partial W} = \frac{1}{N}(\hat{Y} - Y)^{\top}X, \qquad
\frac{\partial L}{\partial b} = \frac{1}{N}(\hat{Y} - Y)^{\top}\mathbf{1}
\]
